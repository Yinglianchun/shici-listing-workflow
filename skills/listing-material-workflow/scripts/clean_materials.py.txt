"""Optional mechanical helper. Run with Python; no packages, network or LLM calls."""
import argparse
import csv
import json
import re
import unicodedata
from decimal import Decimal, InvalidOperation
from pathlib import Path


def normalized(text):
    text=unicodedata.normalize('NFKC',str(text)).casefold()
    text=text.translate(str.maketrans({'’':"'",'‘':"'",'–':'-','—':'-'}))
    return re.sub(r'\s+',' ',text).strip().rstrip('?.!').strip()


def comparison(text):
    return re.sub(r"(?<![\w'])women(?:'s|s)?(?![\w'])",'women',normalized(text))


def ad_text(text,short_question=False):
    """A-column formatting only: preserve hyphens, numeric decimals, and meaning."""
    text=unicodedata.normalize('NFKC',str(text)).translate(str.maketrans({'’':"'",'‘':"'"}))
    if short_question:
        # Only remove question marks. Do not strip other punctuation or question wording.
        return re.sub(r'\s+',' ',text.replace('?','').replace('？','')).strip(),''
    # Expand negative contractions before stripping punctuation.
    text=re.sub(r"\bcan't\b",'cannot',text,flags=re.I)
    text=re.sub(r"\bwon't\b",'will not',text,flags=re.I)
    text=re.sub(r"\bshan't\b",'shall not',text,flags=re.I)
    text=re.sub(r"\b([a-z]+)n't\b",r'\1 not',text,flags=re.I)
    # Preserve the meaning of thousand separators and decimal points.
    text=re.sub(r'(?<![\d.,])\d{1,3}(?:,\d{3})+(?:\.\d+)?(?![\d.,])',lambda m:m[0].replace(',',''),text)
    if re.search(r'\d,\d',text):
        return None,'数值中的逗号含义待确认；需确认部件及单位'
    text=text.translate(str.maketrans({'–':'-','—':'-','‐':'-','‑':'-','−':'-'}))
    result=[]
    for index,char in enumerate(text):
        decimal=(char=='.' and index>0 and index<len(text)-1
                 and text[index-1].isdigit() and text[index+1].isdigit())
        if char=='-' or decimal:
            result.append(char)
        elif unicodedata.category(char).startswith('P'):
            result.append(' ')
        else:
            result.append(char)
    return re.sub(r'\s+',' ',''.join(result)).strip(),''


def strip_brands(text,brands):
    cleaned=str(text);removed=[]
    for brand in brands:
        letters=[re.escape(c) for c in str(brand) if c.isalnum()]
        if not letters:continue
        cleaned,count=re.subn(r'(?<!\w)'+r'[\W_]*'.join(letters)+r'(?!\w)',' ',cleaned,flags=re.I)
        if count:removed.append(str(brand))
    return re.sub(r'\s+',' ',cleaned).strip(),removed


def containment_candidates(entries):
    """Mark contiguous whole-word containment in the same scope; never delete."""
    token_rows=[];index={};candidates=[]
    for position,entry in enumerate(entries):
        tokens=tuple(comparison(entry['text']).split())
        token_rows.append(tokens)
        for token in set(tokens):
            index.setdefault((tuple(entry['scope']),token),set()).add(position)
    for short_pos,short in enumerate(entries):
        tokens=token_rows[short_pos]
        if not tokens:continue
        buckets=[index[(tuple(short['scope']),token)] for token in set(tokens)]
        matches=set.intersection(*buckets)
        for long_pos in sorted(matches):
            long_tokens=token_rows[long_pos]
            if len(long_tokens)<=len(tokens):continue
            if not any(long_tokens[start:start+len(tokens)]==tokens
                       for start in range(len(long_tokens)-len(tokens)+1)):continue
            long=entries[long_pos]
            candidates.append({'short_id':short['id'],'long_id':long['id'],
                'short_text':short['text'],'long_text':long['text'],
                'relation':'ordered_containment','status':'pending_applicability',
                'reason':'同顺序完整包含，仅标记；03确认适用后由04判断，不自动删除'})
    return candidates


def process(rows,mode,brands=None):
    groups={};pending=[]
    for index,source in enumerate(rows,1):
        source=dict(source)
        text=source.get('Keyword')
        if not isinstance(text,str):raise ValueError(f'第{index}行缺少字符串Keyword字段')
        if not text.strip():continue
        kind=source.get('kind') or source.get('Type') or 'unknown'
        kind={'关键词':'keyword','买家问句':'question','ai_generated_question':'question'}.get(kind,kind)
        output=dict(source)
        if mode=='strip-brands':
            cleaned,removed=strip_brands(text,brands or [])
            if not cleaned:
                pending.append({'row':index,'original':source,'reason':'去品牌后没有有效表达，原文保留待核对'});continue
            output['Keyword']=cleaned;output['RemovedBrands']=removed
        if mode=='ad-a':
            original_question=(source.get('原问句') or source.get('OriginalQuestion')
                               or source.get('original_question') or source.get('FullQuestion'))
            # Only an unprocessed original question qualifies for the under-80 exception.
            is_unprocessed_question=(kind=='question'
                                    and (not original_question
                                         or normalized(text)==normalized(original_question)))
            question_length=len(str(original_question or text).strip()) if is_unprocessed_question else 0
            if is_unprocessed_question and question_length>=80:
                pending.append({'row':index,'original':source,
                    'reason':'原问句不少于80字符，须由06提炼投放关键表达并另填描述用清洗表达'});continue
            cleaned,reason=ad_text(text,short_question=(is_unprocessed_question and question_length<80))
            if cleaned is None:
                pending.append({'row':index,'original':source,'reason':reason});continue
            output['Keyword']=cleaned
        identity=normalized(output['Keyword'])
        # Only exact formatting-normalized duplicates merge here. Lexical variants remain for stage 06.
        key=(source.get('sheet',''),source.get('Marketplace',''),kind,identity)
        if key not in groups:
            groups[key]={'row':output,'originals':[],'source_rows':[],'original_order':index,'kind':kind}
        groups[key]['originals'].append(source)
        groups[key]['source_rows'].append(index)
    records=list(groups.values())
    # Sort for reading only. This is not semantic grouping or a deletion rule.
    records.sort(key=lambda g:(g['row'].get('sheet',''),g['kind'],
        tuple(sorted(re.findall(r"[\w'-]+",comparison(g['row']['Keyword'])))),g['original_order']))
    candidates=containment_candidates([{'id':g['original_order'],'text':g['row']['Keyword'],
        'scope':(g['row'].get('sheet',''),g['row'].get('Marketplace',''),g['kind'])}
        for g in records])
    return {'mode':mode,'input_rows':len(rows),'kept_rows':len(records),
        'removed_duplicates':sum(len(g['source_rows'])-1 for g in records),
        'records':records,'pending':pending,'containment_candidates':candidates,
        'semantic_grouping':False}


def extract_cores(rows):
    """Read only explicit core fields; candidates still require owner confirmation."""
    candidates={};found=False
    for row in rows:
        for field,value in row.items():
            if re.sub(r'[\s_-]+','',str(field)).casefold() not in ('coreterm','category','核心词','核心泛词'):continue
            found=True
            if isinstance(value,str):
                value=value.strip()
                if value.startswith('['):
                    try:value=json.loads(value)
                    except ValueError:pass
            for term in value if isinstance(value,list) else [value]:
                if not isinstance(term,str) or not term.strip():continue
                identity=comparison(term)
                item=candidates.setdefault(identity,{'term':term.strip(),'fields':[],'sources':[]})
                if field not in item['fields']:item['fields'].append(field)
                source=row.get('source_file',row.get('file',''))
                if source and source not in item['sources']:item['sources'].append(source)
    return {'mode':'core-terms','input_rows':len(rows),'core_fields_found':found,
            'core_candidates':list(candidates.values()),'confirmed':False}


def sales_terms(rows,term_column='用户搜索词',orders_column='订单数',skip_first_data_row=True):
    """Clean one product's single report period; keep only terms and order counts."""
    groups={};pending=[];counts={'summary_rows':0,'asin_rows':0,'zero_order_rows':0,'empty_terms':0}
    totals={'总订单数','总订单','总计','合计','汇总','grand total','total','total orders'}
    for index,source in enumerate(rows,1):
        if index==1 and skip_first_data_row:
            counts['summary_rows']+=1;continue
        if term_column not in source or orders_column not in source:
            raise ValueError(f'第{index}行缺少{term_column}或{orders_column}列')
        term=str(source[term_column] or '').strip()
        if not term:counts['empty_terms']+=1;continue
        if re.sub(r'[:：]+$','',normalized(term)).strip() in totals:
            counts['summary_rows']+=1;continue
        # A whole ASIN cell, not an embedded code or a ten-letter word such as binoculars.
        if re.fullmatch(r'B[A-Z0-9]{9}',term,re.I) and re.search(r'\d',term):
            counts['asin_rows']+=1;continue
        original={term_column:source[term_column],orders_column:source[orders_column]}
        value=source[orders_column]
        try:
            if value is None or isinstance(value,bool):raise ValueError()
            number=unicodedata.normalize('NFKC',str(value)).strip()
            if re.fullmatch(r'\d{1,3}(?:,\d{3})+(?:\.0+)?',number):number=number.replace(',','')
            orders=Decimal(number)
            if not orders.is_finite() or orders<0 or orders!=orders.to_integral_value():raise ValueError()
            orders=int(orders)
        except (InvalidOperation,ValueError):
            pending.append({'row':index,'original':original,'reason':'订单数不是明确的非负整数，未当成0或参与汇总'});continue
        if orders==0:counts['zero_order_rows']+=1;continue
        identity=normalized(term)
        if identity not in groups:
            groups[identity]={'row':{'Keyword':normalized(term),'Type':'keyword','订单数':0},
                'originals':[],'source_rows':[],'kind':'keyword'}
        record=groups[identity];record['row']['订单数']+=orders
        record['originals'].append(original);record['source_rows'].append(index)
    records=sorted(groups.values(),key=lambda g:(-g['row']['订单数'],g['row']['Keyword']))
    terms=[{'用户搜索词':g['row']['Keyword'],'订单数':g['row']['订单数']} for g in records]
    return {'mode':'sales-terms','input_rows':len(rows),'kept_rows':len(records),
        'removed_duplicates':sum(len(g['source_rows'])-1 for g in records),
        'removed_rows':counts,'terms':terms,'records':records,'pending':pending,
        'total_orders':sum(g['row']['订单数'] for g in records),'semantic_grouping':False}


def main():
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--input',required=True,help='CSV or JSON array with Keyword, optional Type/kind and sheet')
    parser.add_argument('--output',required=True,help='New JSON result; source files are never overwritten')
    parser.add_argument('--mode',choices=['core-terms','sales-terms','dedup','ad-a','strip-brands'],required=True)
    parser.add_argument('--term-column',default='用户搜索词',help='Search-term column for sales-terms mode')
    parser.add_argument('--orders-column',default='订单数',help='Order-count column for sales-terms mode')
    parser.add_argument('--keep-first-row',action='store_true',help='For an already trimmed report without the first summary data row')
    parser.add_argument('--brands',help='JSON array of confirmed third-party brand names; never include own brand')
    args=parser.parse_args();source=Path(args.input);target=Path(args.output)
    if source.resolve()==target.resolve() or target.exists():raise ValueError('输出必须是尚不存在的新文件')
    if source.suffix.lower()=='.csv':
        with source.open(encoding='utf-8-sig',newline='') as stream:rows=list(csv.DictReader(stream))
    else:rows=json.loads(source.read_text(encoding='utf-8-sig'))
    if not isinstance(rows,list):raise ValueError('JSON输入必须是行对象数组')
    brands=json.loads(Path(args.brands).read_text(encoding='utf-8-sig')) if args.brands else []
    if not isinstance(brands,list) or any(not isinstance(v,str) for v in brands):raise ValueError('品牌文件必须是品牌名称字符串数组')
    if args.mode=='strip-brands' and not args.brands:raise ValueError('去品牌模式需要--brands品牌清单')
    if args.mode=='core-terms':result=extract_cores(rows)
    elif args.mode=='sales-terms':result=sales_terms(rows,args.term_column,args.orders_column,not args.keep_first_row)
    else:result=process(rows,args.mode,brands)
    target.parent.mkdir(parents=True,exist_ok=True)
    target.write_text(json.dumps(result,ensure_ascii=False,indent=2),encoding='utf-8')
    print(json.dumps({k:result[k] for k in ['mode','input_rows','kept_rows','removed_duplicates','core_fields_found'] if k in result},ensure_ascii=False))


if __name__=='__main__':main()
